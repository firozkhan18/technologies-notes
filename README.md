**`CompletableFuture`** 

```java 
package com.javacoding.concurrency;

import java.util.concurrent.CompletableFuture;

public class CompletableFutureExample {

    public static void main(String[] args) throws Exception {
    	
//    	1. supplyAsync(Supplier<U> supplier)
//    	Description: This method is used to asynchronously execute a task and return a result. 
//    		         It accepts a Supplier and runs it asynchronously, returning a CompletableFuture<U>.
    	
    	CompletableFuture<Integer> future1 = CompletableFuture.supplyAsync(() -> {
    	    return 5 * 5;
    	});
//    	2. runAsync(Runnable runnable)
//    	Description: This method executes a Runnable asynchronously but does not return a result. 
//    			     It’s useful when you need to perform side-effects (e.g., logging or updating some state).

    	CompletableFuture<Void> future2 = CompletableFuture.runAsync(() -> {
    	    System.out.println("Running task asynchronously.");
    	});
//    	3. thenApply(Function<? super T,? extends U> fn)
//    	Description: This method is used to apply a function to the result of the future when it completes. 
//    	             It transforms the result and returns a new CompletableFuture<U>.

    	CompletableFuture<Integer> result1 = CompletableFuture.supplyAsync(() -> 5)
        .thenApply(value -> value * 2);
        // result will contain 10

//    	4. thenAccept(Consumer<? super T> action)
//    	Description: This method is used when you just want to consume the result of a completed CompletableFuture without changing it. 
//    	             It applies a Consumer to the result.

    	CompletableFuture<Void> result2 = CompletableFuture.supplyAsync(() -> 10)
    		    .thenAccept(value -> System.out.println("Value: " + value));
		// Output: "Value: 10"
    	
//    	5. thenCompose(Function<? super T, ? extends CompletableFuture<U>> fn)
//    	Description: This method allows you to chain two asynchronous operations. 
//    	             Unlike thenApply, which returns a transformed result, thenCompose returns a new CompletableFuture<U>. It's useful for dependent asynchronous tasks.

    	CompletableFuture<Integer> result3 = CompletableFuture.supplyAsync(() -> 5)
    		    .thenCompose(value -> CompletableFuture.supplyAsync(() -> value * 2));
    		// result will contain 10

//    	6. exceptionally(Function<Throwable, ? extends T> fn)
//    	Description: This method allows you to handle exceptions that occur during asynchronous execution. 
//    	             If the CompletableFuture completes exceptionally, it will invoke the provided function to handle the exception and return a fallback value.

    	CompletableFuture<Integer> result4 = CompletableFuture.supplyAsync(() -> {
    	    if (true) throw new RuntimeException("Failure");
    	    return 5;
    	}).exceptionally(ex -> {
    	    System.out.println("Exception: " + ex.getMessage());
    	    return 0;
    	});
    	// Output: Exception: Failure
    	// result will contain 0
		
//    	7. handle(BiFunction<? super T, Throwable, ? extends U> fn)
//    	Description: This method allows you to handle both the result and the exception in one step. 
//    	             It’s similar to exceptionally, but provides both the result and the exception (if any) to the handler.
       
    	CompletableFuture<Integer> result5 = CompletableFuture.supplyAsync(() -> {
    	    if (true) throw new RuntimeException("Failure");
    	    return 5;
    	}).handle((res, ex) -> {
    	    if (ex != null) {
    	        System.out.println("Exception: " + ex.getMessage());
    	        return 0;
    	    }
    	    return res * 2;
    	});
    	// Output: Exception: Failure
    	// result will contain 0

//    	8. whenComplete(BiConsumer<? super T, ? super Throwable> action)
//    	Description: Similar to handle, but doesn't allow you to modify the result. 
//    	             It just lets you observe the completion (whether successful or exceptionally).

    	CompletableFuture<Integer> result6 = CompletableFuture.supplyAsync(() -> 5)
    		    .whenComplete((res, ex) -> {
    		        if (ex != null) {
    		            System.out.println("Exception: " + ex.getMessage());
    		        } else {
    		            System.out.println("Result: " + res);
    		        }
    		    });
    		// Output: Result: 5

//    	9. join()
//    	Description: This method blocks the current thread and waits for the completion of the CompletableFuture. 
//    	             If it completes exceptionally, it throws an exception.

    	Integer result7 = CompletableFuture.supplyAsync(() -> 5).join();
    	// result will contain 5

//    	10. get()
//    	Description: Similar to join(), but throws checked exceptions (ExecutionException or InterruptedException) if something goes wrong during the execution of the future.

    	try {
    	    Integer result = CompletableFuture.supplyAsync(() -> 10).get();
    	    // result will contain 10
    	} catch (Exception e) {
    	    e.printStackTrace();
    	}

//    	11. allOf(CompletableFuture<?>... cfs)
//    	Description: This method takes an array of CompletableFuture instances and returns a new CompletableFuture<Void>. 
//    				This future will complete when all the given futures complete.
    	
    	CompletableFuture<Void> allOf = CompletableFuture.allOf(
    		    CompletableFuture.supplyAsync(() -> 5),
    		    CompletableFuture.supplyAsync(() -> 10)
    		);
    		allOf.join();

//    	12. anyOf(CompletableFuture<?>... cfs)
//    	Description: This method takes an array of CompletableFuture instances and returns a new CompletableFuture<Object>. 
//    	             This future will complete when any one of the given futures completes.
    	
    		
		CompletableFuture<Object> anyOf = CompletableFuture.anyOf(
			    CompletableFuture.supplyAsync(() -> 5),
			    CompletableFuture.supplyAsync(() -> 10)
			);
		System.out.println(anyOf.join()); // Prints either 5 or 10
	
    		// Supply async with a computation
        CompletableFuture<Integer> future0 = CompletableFuture.supplyAsync(() -> 2);
        
        // Chain another computation using thenApply
        CompletableFuture<Integer> future00 = future0.thenApply(result0 -> result0 * 5);
        
        // Handle exceptions using exceptionally
        CompletableFuture<Integer> future000 = future00.exceptionally(ex -> {
            System.out.println("Exception: " + ex.getMessage());
            return 0;
        });

        // Wait for the final result
        System.out.println("Final Result: " + future000.join());
        
        
        CompletableFuture<Integer> future11 = CompletableFuture.supplyAsync(() -> 5);
        CompletableFuture<Integer> future12 = CompletableFuture.supplyAsync(() -> {
            if (true) throw new RuntimeException("Error in future2");
            return 10;
        });

        CompletableFuture<Void> combined = CompletableFuture.allOf(future11, future12)
            .handle((result, ex) -> {
                if (ex != null) {
                    System.out.println("Exception: " + ex.getMessage());
                } else {
                    System.out.println("Both futures completed");
                }
                return null;
            });

        combined.join(); // Wait for both futures to complete
        
//        exceptionally(): Handles exceptions and provides a fallback value.
//        handle(): Handles both the result and exception, and can modify the result.
//        whenComplete(): Handles side-effects like logging, but doesn't modify the result.
//        thenCombine(): Combines the results of two CompletableFuture instances.
//        allOf(): Waits for all CompletableFuture instances to complete.
//        anyOf(): Waits for any one CompletableFuture to complete.
    }

}
```

### 1. **Basic CompletableFuture Usage**
**Question**: Write a program that fetches data asynchronously using `CompletableFuture`. 
- Simulate fetching user details by calling a method `fetchUserDetails()`, which returns user data after a delay.
- Use `CompletableFuture` to call `fetchUserDetails()` asynchronously, and print the user data once the fetch is complete.

```java
import java.util.concurrent.CompletableFuture;
import java.util.concurrent.TimeUnit;

public class CompletableFutureExample {
    public static void main(String[] args) throws Exception {
        CompletableFuture<Void> future = CompletableFuture.runAsync(() -> {
            try {
                // Simulate delay
                TimeUnit.SECONDS.sleep(2);
                System.out.println("Fetching user details...");
            } catch (InterruptedException e) {
                e.printStackTrace();
            }
        });

        // Block and wait for the asynchronous task to complete
        future.get();
    }
}
```

### 2. **Chaining Asynchronous Tasks**
**Question**: Write a program that executes two tasks asynchronously:
1. The first task fetches user details (`fetchUserDetails()`).
2. The second task processes the fetched user data (`processUserData()`).

Use `thenCompose` to chain these tasks in an asynchronous, non-blocking manner.

```java
import java.util.concurrent.CompletableFuture;
import java.util.concurrent.TimeUnit;

public class CompletableFutureChaining {
    public static void main(String[] args) throws Exception {
        CompletableFuture<String> fetchData = CompletableFuture.supplyAsync(() -> {
            try {
                TimeUnit.SECONDS.sleep(2);
                return "User Data";
            } catch (InterruptedException e) {
                throw new RuntimeException(e);
            }
        });

        CompletableFuture<String> processData = fetchData.thenCompose(data -> {
            return CompletableFuture.supplyAsync(() -> {
                return "Processed " + data;
            });
        });

        // Block and get the result
        System.out.println(processData.get());
    }
}
```

### 3. **Handling Exceptions with CompletableFuture**
**Question**: Write a program that simulates an asynchronous task that may throw an exception. Use `handle` or `exceptionally` to handle the exception gracefully.

```java
import java.util.concurrent.CompletableFuture;
import java.util.concurrent.ExecutionException;
import java.util.concurrent.TimeUnit;

public class CompletableFutureExceptionHandling {
    public static void main(String[] args) throws ExecutionException, InterruptedException {
        CompletableFuture<String> future = CompletableFuture.supplyAsync(() -> {
            try {
                TimeUnit.SECONDS.sleep(2);
                throw new RuntimeException("Failed to fetch data");
            } catch (InterruptedException e) {
                throw new RuntimeException(e);
            }
        });

        // Handling exceptions using exceptionally
        CompletableFuture<String> result = future.exceptionally(ex -> {
            return "Fallback data: " + ex.getMessage();
        });

        // Block and get the result
        System.out.println(result.get());
    }
}
```

### 4. **Combining Multiple CompletableFutures**
**Question**: Write a program that runs multiple tasks asynchronously, and combine their results using `thenCombine` or `allOf`.
- Task 1: Fetch user details
- Task 2: Fetch user orders

Combine the results once both tasks are complete.

```java
import java.util.concurrent.CompletableFuture;
import java.util.concurrent.TimeUnit;

public class CompletableFutureCombine {
    public static void main(String[] args) throws Exception {
        CompletableFuture<String> fetchUserDetails = CompletableFuture.supplyAsync(() -> {
            try {
                TimeUnit.SECONDS.sleep(2);
                return "User Details";
            } catch (InterruptedException e) {
                throw new RuntimeException(e);
            }
        });

        CompletableFuture<String> fetchUserOrders = CompletableFuture.supplyAsync(() -> {
            try {
                TimeUnit.SECONDS.sleep(1);
                return "User Orders";
            } catch (InterruptedException e) {
                throw new RuntimeException(e);
            }
        });

        // Combine both results
        CompletableFuture<String> combinedResult = fetchUserDetails
            .thenCombine(fetchUserOrders, (userDetails, userOrders) -> userDetails + " + " + userOrders);

        System.out.println(combinedResult.get()); // Output: User Details + User Orders
    }
}
```

### 5. **AllOf and AnyOf**
**Question 1**: Write a program that runs multiple independent asynchronous tasks using `CompletableFuture.allOf()`. Wait for all tasks to complete, and then combine their results.

```java
import java.util.concurrent.CompletableFuture;
import java.util.concurrent.ExecutionException;
import java.util.concurrent.TimeUnit;

public class CompletableFutureAllOf {
    public static void main(String[] args) throws ExecutionException, InterruptedException {
        CompletableFuture<String> task1 = CompletableFuture.supplyAsync(() -> {
            try {
                TimeUnit.SECONDS.sleep(2);
                return "Task 1 completed";
            } catch (InterruptedException e) {
                return "Task 1 failed";
            }
        });

        CompletableFuture<String> task2 = CompletableFuture.supplyAsync(() -> {
            try {
                TimeUnit.SECONDS.sleep(1);
                return "Task 2 completed";
            } catch (InterruptedException e) {
                return "Task 2 failed";
            }
        });

        CompletableFuture<Void> allOf = CompletableFuture.allOf(task1, task2);
        allOf.get();  // Block until all tasks complete

        // Combine results after all tasks complete
        System.out.println(task1.get());
        System.out.println(task2.get());
    }
}
```

### 6. **Timeout Handling with CompletableFuture**
**Question**: Write a program that fetches data asynchronously with a timeout. Use `completeOnTimeout()` to provide a fallback value if the task exceeds the timeout.

```java
import java.util.concurrent.CompletableFuture;
import java.util.concurrent.TimeUnit;

public class CompletableFutureTimeout {
    public static void main(String[] args) throws Exception {
        CompletableFuture<String> future = CompletableFuture.supplyAsync(() -> {
            try {
                TimeUnit.SECONDS.sleep(5); // Simulating a long task
                return "Fetched Data";
            } catch (InterruptedException e) {
                return "Failed to fetch data";
            }
        });

        // Set a timeout of 2 seconds
        CompletableFuture<String> result = future.completeOnTimeout("Fallback data", 2, TimeUnit.SECONDS);

        System.out.println(result.get()); // Output: Fallback data
    }
}
```

### 7. **Custom Executor with CompletableFuture**
**Question**: Write a program that creates a custom executor to run tasks asynchronously using `CompletableFuture`.

```java
import java.util.concurrent.CompletableFuture;
import java.util.concurrent.Executors;
import java.util.concurrent.ExecutorService;
import java.util.concurrent.TimeUnit;

public class CompletableFutureWithCustomExecutor {
    public static void main(String[] args) throws Exception {
        ExecutorService executor = Executors.newFixedThreadPool(2);

        CompletableFuture<String> future = CompletableFuture.supplyAsync(() -> {
            try {
                TimeUnit.SECONDS.sleep(2);
                return "Completed Task";
            } catch (InterruptedException e) {
                return "Error";
            }
        }, executor);

        // Block and get the result
        System.out.println(future.get());
        executor.shutdown();
    }
}
```
